cmake_minimum_required(VERSION 3.10)
project(sony_lut LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_library(jpeg STATIC IMPORTED)
set(JPEG_BUILD "${ROOT}/out/jpeg-android" CACHE PATH "Cross-compiled libjpeg directory")
set_target_properties(jpeg PROPERTIES IMPORTED_LOCATION "${JPEG_BUILD}/libjpeg.a")
include_directories("${ROOT}/vendor/libjpeg-turbo" "${JPEG_BUILD}")
add_library(sonylut SHARED core.cpp lut_cache.cpp jpeg_io.cpp jni.cpp device_memory.cpp uyvy.cpp preview_fit.cpp)
target_compile_options(sonylut PRIVATE -O3 -fno-vectorize -fno-slp-vectorize)
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  # On the A6300 Cortex-A5, ARM code beats Thumb for the exact packed LUT loop.
  target_compile_options(sonylut PRIVATE -marm)
endif()
target_link_libraries(sonylut jpeg jnigraphics log dl)
add_executable(sony-lut core.cpp jpeg_io.cpp cli.cpp)
target_compile_options(sony-lut PRIVATE -O3 -fno-vectorize -fno-slp-vectorize)
target_link_libraries(sony-lut jpeg)
option(SONY_BUILD_BENCHMARKS "Build CPU-only UYVY benchmark" OFF)
if(SONY_BUILD_BENCHMARKS)
  add_executable(sony-cache-tests core.cpp uyvy.cpp uyvy_cache_tests.cpp)
  target_compile_options(sony-cache-tests PRIVATE -O3 -fno-vectorize -fno-slp-vectorize)
  foreach(mode thumb arm)
    add_executable(sony-uyvy-bench-${mode} core.cpp uyvy.cpp uyvy_bench.cpp)
    target_compile_options(sony-uyvy-bench-${mode} PRIVATE -O3 -fno-vectorize -fno-slp-vectorize -m${mode})
    target_link_libraries(sony-uyvy-bench-${mode} jpeg)
  endforeach()
endif()
